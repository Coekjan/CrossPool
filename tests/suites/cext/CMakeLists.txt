include(GoogleTest)

set(XPOOL_CEXT_TEST_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
file(GLOB_RECURSE XPOOL_PUBLIC_HEADERS CONFIGURE_DEPENDS
  "${PROJECT_SOURCE_DIR}/src/cext-include/xpool/*.hpp"
  "${PROJECT_SOURCE_DIR}/src/cext-include/xpool/*.cuh"
)
set(XPOOL_HEADER_COMPILE_DIR "${CMAKE_CURRENT_BINARY_DIR}/header_compile")
file(MAKE_DIRECTORY "${XPOOL_HEADER_COMPILE_DIR}")
set(XPOOL_HEADER_COMPILE_SOURCES)
foreach(XPOOL_PUBLIC_HEADER IN LISTS XPOOL_PUBLIC_HEADERS)
  file(RELATIVE_PATH XPOOL_PUBLIC_HEADER_REL "${PROJECT_SOURCE_DIR}/src/cext-include" "${XPOOL_PUBLIC_HEADER}")
  string(REPLACE "/" "_" XPOOL_HEADER_COMPILE_STEM "${XPOOL_PUBLIC_HEADER_REL}")
  string(REPLACE "." "_" XPOOL_HEADER_COMPILE_STEM "${XPOOL_HEADER_COMPILE_STEM}")
  if(XPOOL_PUBLIC_HEADER MATCHES "\\.cuh$")
    set(XPOOL_HEADER_COMPILE_EXTENSION "cu")
  else()
    set(XPOOL_HEADER_COMPILE_EXTENSION "cpp")
  endif()
  set(XPOOL_HEADER_COMPILE_SOURCE
      "${XPOOL_HEADER_COMPILE_DIR}/${XPOOL_HEADER_COMPILE_STEM}.${XPOOL_HEADER_COMPILE_EXTENSION}")
  file(CONFIGURE OUTPUT "${XPOOL_HEADER_COMPILE_SOURCE}"
       CONTENT "#include <${XPOOL_PUBLIC_HEADER_REL}>\n")
  list(APPEND XPOOL_HEADER_COMPILE_SOURCES "${XPOOL_HEADER_COMPILE_SOURCE}")
endforeach()

add_library(xpool_cext_header_compile OBJECT ${XPOOL_HEADER_COMPILE_SOURCES})
target_link_libraries(xpool_cext_header_compile
  PRIVATE
    xpool_abi
    xpool_compiler_options
    NVSHMEM::NVSHMEM
    ${TORCH_LIBRARIES}
    CUDA::cudart
)
target_compile_features(xpool_cext_header_compile PRIVATE cxx_std_20)
set_target_properties(xpool_cext_header_compile PROPERTIES
  CUDA_ARCHITECTURES "${XPOOL_EFFECTIVE_CUDA_ARCHITECTURES}"
  CUDA_STANDARD 20
  CUDA_STANDARD_REQUIRED ON
  CUDA_SEPARABLE_COMPILATION ON
)

file(GLOB_RECURSE XPOOL_CEXT_TEST_SOURCES CONFIGURE_DEPENDS
  "${XPOOL_CEXT_TEST_DIR}/*_test.c"
  "${XPOOL_CEXT_TEST_DIR}/*_test.cc"
  "${XPOOL_CEXT_TEST_DIR}/*_test.cpp"
  "${XPOOL_CEXT_TEST_DIR}/*_test.cu"
)

if(NOT XPOOL_CEXT_TEST_SOURCES)
  message(FATAL_ERROR "XPOOL_BUILD_CEXT_TESTS is ON but no csrc tests were found under ${XPOOL_CEXT_TEST_DIR}")
endif()

set(XPOOL_CEXT_TEST_TARGETS)
foreach(XPOOL_CEXT_TEST_SOURCE IN LISTS XPOOL_CEXT_TEST_SOURCES)
  file(RELATIVE_PATH XPOOL_CEXT_TEST_REL "${XPOOL_CEXT_TEST_DIR}" "${XPOOL_CEXT_TEST_SOURCE}")
  string(REGEX REPLACE "\\.(c|cc|cpp|cu)$" "" XPOOL_CEXT_TEST_STEM "${XPOOL_CEXT_TEST_REL}")
  string(REPLACE "/" "_" XPOOL_CEXT_TEST_TARGET_SUFFIX "${XPOOL_CEXT_TEST_STEM}")
  string(REGEX REPLACE "_test$" "" XPOOL_CEXT_TEST_NAME_STEM "${XPOOL_CEXT_TEST_STEM}")
  string(REPLACE "/" "." XPOOL_CEXT_TEST_NAME_SUFFIX "${XPOOL_CEXT_TEST_NAME_STEM}")

  set(XPOOL_CEXT_TEST_TARGET "xpool_cext_${XPOOL_CEXT_TEST_TARGET_SUFFIX}")
  add_executable("${XPOOL_CEXT_TEST_TARGET}" "${XPOOL_CEXT_TEST_SOURCE}")
  list(APPEND XPOOL_CEXT_TEST_TARGETS "${XPOOL_CEXT_TEST_TARGET}")
  target_link_libraries(
    "${XPOOL_CEXT_TEST_TARGET}"
    PRIVATE
      xpool_abi
      xpool_compiler_options
      xpool_cext_core
      NVSHMEM::NVSHMEM
      ${TORCH_LIBRARIES}
      CUDA::cudart
      GTest::gtest_main
  )
  target_include_directories("${XPOOL_CEXT_TEST_TARGET}" PRIVATE "${XPOOL_CEXT_TEST_DIR}")
  target_compile_features("${XPOOL_CEXT_TEST_TARGET}" PRIVATE cxx_std_20)
  set_target_properties("${XPOOL_CEXT_TEST_TARGET}" PROPERTIES
    CUDA_ARCHITECTURES "${XPOOL_EFFECTIVE_CUDA_ARCHITECTURES}"
    CUDA_STANDARD 20
    CUDA_STANDARD_REQUIRED ON
    TEST_LAUNCHER
      "${Python3_EXECUTABLE};${PROJECT_SOURCE_DIR}/tests/harness/runner/ctest_launcher.py"
  )
  if(XPOOL_CEXT_TEST_SOURCE MATCHES "\\.cu$")
    set_property(TARGET "${XPOOL_CEXT_TEST_TARGET}"
      PROPERTY CUDA_SEPARABLE_COMPILATION ON)
  endif()

  set(XPOOL_CEXT_TEST_PROPERTIES TIMEOUT 20)
  if(XPOOL_CEXT_TEST_SOURCE MATCHES "\\.cu$")
    list(APPEND XPOOL_CEXT_TEST_PROPERTIES RESOURCE_GROUPS "gpus:1")
  endif()
  gtest_discover_tests("${XPOOL_CEXT_TEST_TARGET}"
    TEST_PREFIX "cext.${XPOOL_CEXT_TEST_NAME_SUFFIX}."
    DISCOVERY_MODE PRE_TEST
    DISCOVERY_TIMEOUT 30
    PROPERTIES ${XPOOL_CEXT_TEST_PROPERTIES}
  )
endforeach()

add_custom_target(xpool_cext_tests DEPENDS xpool_cext_header_compile ${XPOOL_CEXT_TEST_TARGETS})
